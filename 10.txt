a) SCAN – Elevator: go one direction, hit boundary, then reverse
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

int main() {
    int n, head, disk_size, direction;

    cout << "Enter number of disk requests: ";
    cin >> n;

    vector<int> requests(n);

    cout << "Enter the disk request sequence:\n";
    for (int i = 0; i < n; i++) {
        cin >> requests[i];
    }

    cout << "Enter initial head position: ";
    cin >> head;

    cout << "Enter disk size (total cylinders): ";
    cin >> disk_size;

    cout << "Enter direction (0 = left, 1 = right): ";
    cin >> direction;

    vector<int> left, right;

    // Divide requests into left and right of head
    for (int i = 0; i < n; i++) {
        if (requests[i] < head)
            left.push_back(requests[i]);
        else
            right.push_back(requests[i]);
    }

    // Add end points
    if (direction == 0)
        left.push_back(0);
    else
        right.push_back(disk_size - 1);

    // Sort both sides
    sort(left.begin(), left.end());
    sort(right.begin(), right.end());

    int totalSeekTime = 0;
    cout << "\n--- SCAN Disk Scheduling ---\n";
    cout << "Seek Sequence: " << head;

    if (direction == 1) {
        // Move right first
        for (int i = 0; i < right.size(); i++) {
            cout << " -> " << right[i];
            totalSeekTime += abs(right[i] - head);
            head = right[i];
        }

        // Then reverse to left
        for (int i = left.size() - 1; i >= 0; i--) {
            cout << " -> " << left[i];
            totalSeekTime += abs(left[i] - head);
            head = left[i];
        }
    } else {
        // Move left first
        for (int i = left.size() - 1; i >= 0; i--) {
            cout << " -> " << left[i];
            totalSeekTime += abs(left[i] - head);
            head = left[i];
        }

        // Then reverse to right
        for (int i = 0; i < right.size(); i++) {
            cout << " -> " << right[i];
            totalSeekTime += abs(right[i] - head);
            head = right[i];
        }
    }

    cout << "\nTotal Seek Time = " << totalSeekTime << endl;

    return 0;
}

b) C-SCAN – Go right to end, jump to 0 (no service), then serve left
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

int main() {
    int n, head, disk_size;

    cout << "Enter number of disk requests: ";
    cin >> n;

    vector<int> requests(n);

    cout << "Enter the disk request sequence:\n";
    for (int i = 0; i < n; i++) {
        cin >> requests[i];
    }

    cout << "Enter initial head position: ";
    cin >> head;

    cout << "Enter disk size (total cylinders): ";
    cin >> disk_size;

    vector<int> left, right;

    // Divide requests into left and right of head
    for (int i = 0; i < n; i++) {
        if (requests[i] < head)
            left.push_back(requests[i]);
        else
            right.push_back(requests[i]);
    }

    // Add end points (for circular movement)
    left.push_back(0);
    right.push_back(disk_size - 1);

    // Sort both sides
    sort(left.begin(), left.end());
    sort(right.begin(), right.end());

    int totalSeekTime = 0;

    cout << "\n--- C-SCAN Disk Scheduling ---\n";
    cout << "Seek Sequence: " << head;

    // Move towards right first
    for (int i = 0; i < right.size(); i++) {
        cout << " -> " << right[i];
        totalSeekTime += abs(right[i] - head);
        head = right[i];
    }

    // Jump from end to beginning (circular move)
    head = 0;
    cout << " -> " << head;

    // Now service left side
    for (int i = 0; i < left.size(); i++) {
        cout << " -> " << left[i];
        totalSeekTime += abs(left[i] - head);
        head = left[i];
    }

    cout << "\nTotal Seek Time = " << totalSeekTime << endl;

    return 0;
}
