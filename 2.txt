a) fork() – Creates a child process
#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>

int main() {
    pid_t pid;

    pid = fork();   // Create a new process

    if (pid < 0) {
        // Error occurred
        printf("Fork failed\n");
    }
    else if (pid == 0) {
        // Child process
        printf("Child Process\n");
        printf("Child PID: %d\n", getpid());
        printf("Parent PID: %d\n", getppid());
    }
    else {
        // Parent process
        printf("Parent Process\n");
        printf("Parent PID: %d\n", getpid());
        printf("Child PID: %d\n", pid);
    }

    return 0;
}





b) exec() – Child replaces itself with another program (ls -l)
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>

int main()
{
    pid_t pid;

    pid = fork();   // Create child process

    if (pid < 0)
    {
        printf("Fork failed\n");
        exit(1);
    }
    else if (pid == 0)
    {
        // Child process
        printf("Child process (PID: %d)\n", getpid());
        printf("Executing ls command using exec...\n");

        execl("/bin/ls", "ls", "-l", NULL);

        // If exec fails
        perror("exec failed");
        exit(1);
    }
    else
    {
        // Parent process
        wait(NULL);   // Wait for child to complete
        printf("Parent process (PID: %d)\n", getpid());
        printf("Child execution completed\n");
    }

    return 0;
}

        exit(1);
    }
    else if (pid == 0) {
        printf("Child Process\n");
        printf("Child PID: %d\n", getpid());
        printf("Parent PID: %d\n", getppid());

        sleep(2);

        printf("Child process exiting\n");
        exit(0);
    }
    else {
        printf("Parent Process\n");
        printf("Parent PID: %d\n", getpid());

        wait(&status);

        printf("Parent resumed after child termination\n");
        printf("Parent process exiting\n");
    }

    return 0;
}


Assignment 4: Reader-Writer Problem
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>
#include <chrono>
using namespace std;

mutex mtx;        
mutex wrt;        
mutex printMtx;   

int readCount = 0;

void reader(int id)
{
   
    {
        lock_guard<mutex> lock(mtx);
        readCount++;
        if (readCount == 1)
            wrt.lock();   
    }


    {
        lock_guard<mutex> lock(printMtx);
        cout << "Reader " << id << " is reading\n";
    }
    this_thread::sleep_for(chrono::seconds(1));

   
    {
        lock_guard<mutex> lock(mtx);
        readCount--;
        if (readCount == 0)
            wrt.unlock(); 
    }
}


void writer(int id)
{
    wrt.lock();

    
    {
        lock_guard<mutex> lock(printMtx);
        cout << "Writer " << id << " is writing\n";
    }
    this_thread::sleep_for(chrono::seconds(2));

    wrt.unlock();
}

int main()
{
    int numReaders = 5;
    int numWriters = 3;

    vector<thread> r, w;

    
    for (int i = 1; i <= numReaders; i++)
        r.push_back(thread(reader, i));

    
    for (int i = 1; i <= numWriters; i++)
        w.push_back(thread(writer, i));

  
    for (auto &t : r) t.join();
    for (auto &t : w) t.join();

    return 0;
}



Assignment 5: Producer-Consumer Problem
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>

using namespace std;

const int BUFFER_SIZE = 5;
queue<int> buffer;

mutex mtx;
condition_variable cv;

void producer() {
    for (int i = 1; i <= 10; i++) {
        unique_lock<mutex> lock(mtx);

        // Wait if buffer is full
        cv.wait(lock, [] { return buffer.size() < BUFFER_SIZE; });

        buffer.push(i);
        cout << "Produced: " << i << endl;

        lock.unlock();
        cv.notify_all();
    }
}

void consumer() {
    for (int i = 1; i <= 10; i++) {
        unique_lock<mutex> lock(mtx);

        // Wait if buffer is empty
        cv.wait(lock, [] { return !buffer.empty(); });

        int item = buffer.front();
        buffer.pop();
        cout << "Consumed: " << item << endl;

        lock.unlock();
        cv.notify_all();
    }
}

int main() {
    thread t1(producer);
    thread t2(consumer);

    t1.join();
    t2.join();

    return 0;
}






Assignment 6: Banker's Algorithm (Deadlock Avoidance)
#include <iostream>
using namespace std;

int main()
{
    int n, m;  

    cout << "Enter number of processes: ";
    cin >> n;

    cout << "Enter number of resource types: ";
    cin >> m;

    int alloc[n][m], max[n][m], avail[m];
    int need[n][m];
    bool finish[n];
    int safeSeq[n];

    // Input Allocation Matrix
    cout << "\nEnter Allocation Matrix:\n";
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            cin >> alloc[i][j];

    // Input Maximum Matrix
    cout << "\nEnter Maximum Matrix:\n";
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            cin >> max[i][j];

    // Input Available Resources
    cout << "\nEnter Available Resources:\n";
    for(int j = 0; j < m; j++)
        cin >> avail[j];

    // Calculate Need Matrix = Max - Allocation
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    // Initialize finish array
    for(int i = 0; i < n; i++)
        finish[i] = false;

    int count = 0;

    while(count < n)
    {
        bool found = false;

        for(int i = 0; i < n; i++)
        {
            if(finish[i] == false)
            {
                int j;
                for(j = 0; j < m; j++)
                {
                    if(need[i][j] > avail[j])
                        break;
                }

                if(j == m)  
                {
                    for(int k = 0; k < m; k++)
                        avail[k] += alloc[i][k];

                    safeSeq[count++] = i;
                    finish[i] = true;
                    found = true;
                }
            }
        }

        if(found == false)
        {
            cout << "\nSystem is NOT in Safe State (Deadlock Possible)\n";
            return 0;
        }
    }

    cout << "\nSystem is in Safe State.\nSafe Sequence is: ";
    for(int i = 0; i < n; i++)
        cout << "P" << safeSeq[i] << " ";

    cout << endl;

    return 0;
}



7. Write a program to simulate memory placement strategies-best fit, first fit,
next fit and worst fit.

//BEST FIT

#include <iostream>
using namespace std;

int main() {
    int nb, np;

    cout << "Enter number of memory blocks: ";
    cin >> nb;

    int blockSize[nb];
    cout << "Enter sizes of memory blocks:\n";
    for (int i = 0; i < nb; i++) {
        cin >> blockSize[i];
    }

    cout << "Enter number of processes: ";
    cin >> np;

    int processSize[np];
    cout << "Enter sizes of processes:\n";
    for (int i = 0; i < np; i++) {
        cin >> processSize[i];
    }

    int allocation[np];
    for (int i = 0; i < np; i++)
        allocation[i] = -1;

    // Best Fit Allocation
    for (int i = 0; i < np; i++) {
        int bestIdx = -1;

        for (int j = 0; j < nb; j++) {
            if (blockSize[j] >= processSize[i]) {
                if (bestIdx == -1 || blockSize[j] < blockSize[bestIdx]) {
                    bestIdx = j;
                }
            }
        }

        if (bestIdx != -1) {
            allocation[i] = bestIdx;
            blockSize[bestIdx] -= processSize[i];
        }
    }

    // Output
    cout << "\nProcess No.\tProcess Size\tBlock No.\n";
    for (int i = 0; i < np; i++) {
        cout << i + 1 << "\t\t" << processSize[i] << "\t\t";
        if (allocation[i] != -1)
            cout << allocation[i] + 1;
        else
            cout << "Not Allocated";
        cout << endl;
    }

    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int nb, np;

    cout << "Enter number of memory blocks: ";
    cin >> nb;

    int blockSize[nb];
    cout << "Enter sizes of memory blocks:\n";
    for (int i = 0; i < nb; i++) {
        cin >> blockSize[i];
    }

    cout << "Enter number of processes: ";
    cin >> np;

    int processSize[np];
    cout << "Enter sizes of processes:\n";
    for (int i = 0; i < np; i++) {
        cin >> processSize[i];
    }

    int allocation[np];
    for (int i = 0; i < np; i++)
        allocation[i] = -1;

    int last_alloc = 0; // pointer to track last allocation

    // Next Fit Allocation
    for (int i = 0; i < np; i++) {
        int count = 0;

        while (count < nb) {
            if (blockSize[last_alloc] >= processSize[i]) {
                allocation[i] = last_alloc;
                blockSize[last_alloc] -= processSize[i];
                break;
            }

            last_alloc = (last_alloc + 1) % nb;
            count++;
        }
    }

    // Output
    cout << "\nProcess No.\tProcess Size\tBlock No.\n";
    for (int i = 0; i < np; i++) {
        cout << i + 1 << "\t\t" << processSize[i] << "\t\t";
        if (allocation[i] != -1)
            cout << allocation[i] + 1;
        else
            cout << "Not Allocated";
        cout << endl;
    }

    return 0;
}
