//FIFO



#include <iostream>
using namespace std;

int main() {
    int pages[50], frames[10];
    int n, f, i, j, k, flag;
    int hit = 0, miss = 0, index = 0;

    cout << "Enter number of pages: ";
    cin >> n;

    cout << "Enter page reference string: ";
    for(i = 0; i < n; i++) {
        cin >> pages[i];
    }

    cout << "Enter number of frames: ";
    cin >> f;

    // Initialize frames with -1
    for(i = 0; i < f; i++) {
        frames[i] = -1;
    }

    cout << "\nPage\tFrames\n";

    for(i = 0; i < n; i++) {
        flag = 0;

        // Check for hit
        for(j = 0; j < f; j++) {
            if(frames[j] == pages[i]) {
                hit++;
                flag = 1;
                break;
            }
        }

        // If miss, replace using FIFO
        if(flag == 0) {
            frames[index] = pages[i];
            index = (index + 1) % f;
            miss++;
        }

        // Display frames
        cout << pages[i] << "\t";
        for(k = 0; k < f; k++) {
            if(frames[k] != -1)
                cout << frames[k] << " ";
            else
                cout << "- ";
        }
        cout << endl;
    }

    cout << "\nTotal Hits = " << hit;
    cout << "\nTotal Misses = " << miss;

    float hit_ratio = (float)hit / n;
    float miss_ratio = (float)miss / n;

    cout << "\nHit Ratio = " << hit_ratio;
    cout << "\nMiss Ratio = " << miss_ratio;

    return 0;
}

Assignment 9: Disk Scheduling – FCFS & SSTF

a) FCFS – Serve requests in arrival order
#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    int n, head;
    
    cout << "Enter number of disk requests: ";
    cin >> n;

    vector<int> requests(n);

    cout << "Enter the disk request sequence:\n";
    for (int i = 0; i < n; i++) {
        cin >> requests[i];
    }

    cout << "Enter initial head position: ";
    cin >> head;

    int totalSeekTime = 0;

    cout << "\n--- FCFS Disk Scheduling ---\n";
    cout << "Seek Sequence: " << head;

    for (int i = 0; i < n; i++) {
        cout << " -> " << requests[i];
        totalSeekTime += abs(requests[i] - head);
        head = requests[i];
    }

    cout << "\nTotal Seek Time = " << totalSeekTime << endl;

    return 0;
}

b) SSTF – Serve closest request first (greedy, mark visited)
#include <iostream>
#include <vector>
#include <cmath>
#include <limits>
using namespace std;

int main() {
    int n, head;

    cout << "Enter number of disk requests: ";
    cin >> n;

    vector<int> requests(n);
    vector<bool> visited(n, false);

    cout << "Enter the disk request sequence:\n";
    for (int i = 0; i < n; i++) {
        cin >> requests[i];
    }

    cout << "Enter initial head position: ";
    cin >> head;

    int totalSeekTime = 0;

    cout << "\n--- SSTF Disk Scheduling ---\n";
    cout << "Seek Sequence: " << head;

    for (int i = 0; i < n; i++) {
        int minDist = numeric_limits<int>::max();
        int index = -1;


        for (int j = 0; j < n; j++) {
            if (!visited[j]) {
                int dist = abs(requests[j] - head);
                if (dist < minDist) {
                    minDist = dist;
                    index = j;
                }
            }
        }

        visited[index] = true;
        totalSeekTime += minDist;
        head = requests[index];

        cout << " -> " << head;
    }

    cout << "\nTotal Seek Time = " << totalSeekTime << endl;

    return 0;
}
